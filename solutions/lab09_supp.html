1.4 Summarizing Accuracy, Precision, Recall, and F1 Measures

The intended output is:

# Accuracy
print(f"Accuracy: {accuracy(gold, pred):.3f}")

print()

# Print header
print(f"{'Tag':<6}{'TP':>4}{'FP':>4}{'FN':>4}{'P':>8}{'R':>8}{'F1':>8}")

# Calculate precision, recall, and F1 for each tag in the gold and precision tags
for tag in sorted(set(gold) | set(pred)):
    # get tp, fp, fn for a given tag
    tp, fp, fn = count_tp_fp_fn(gold, pred, tag)
    # get precision, recall, and f1 measures for a given tag
    print(f"{tag:<6}{tp:>4}{fp:>4}{fn:>4}"
            f"{precision(gold, pred, tag):>8.3f}"
            f"{recall(gold, pred, tag):>8.3f}"
            f"{f1(gold, pred, tag):>8.3f}")
